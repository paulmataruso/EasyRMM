function Write-Result($ok, $data, $code, $message) {
    $r = [ordered]@{ ok = $ok; version = $version; data = $data; error = $null }
    if (-not $ok) { $r.error = [ordered]@{ code = $code; message = $message } }
    [Console]::Out.WriteLine('EASYRMM-WINMOD-RESULT-V1 ' + (ConvertTo-Json -InputObject $r -Compress -Depth 8))
}
function Stop-With($code, $message) {
    Write-Result $false $null $code $message
    exit 0
}
function Test-Param($name) {
    return ($null -ne $p.PSObject.Properties[$name])
}
function Get-Prop($o, $name) {
    if ($null -ne $o -and $null -ne $o.PSObject.Properties[$name]) { return $o.$name }
    return $null
}
function Get-List($o, $name) {
    $v = Get-Prop $o $name
    if ($null -eq $v) { return @() }
    return @($v)
}
function Get-Text($o, $name) {
    $v = Get-Prop $o $name
    if ($null -eq $v) { return '' }
    return [string]$v
}
function Get-UtcText($v) {
    if ($null -eq $v) { return $null }
    return ([datetime]$v).ToUniversalTime().ToString('yyyy-MM-ddTHH:mm:ssZ')
}
function Get-First($o, $name) {
    $v = Get-Prop $o $name
    if ($null -eq $v) { return '' }
    $a = @($v)
    if ($a.Count -eq 0) { return '' }
    return [string]$a[0]
}
function Join-Text([string]$Left, [string]$Right) { return $Left + $Right }
function Test-UnderDn([string]$Dn, [string]$Root) {
    return ($Dn.Equals($Root, 'OrdinalIgnoreCase') -or $Dn.EndsWith(',' + $Root, 'OrdinalIgnoreCase'))
}
function Test-ServiceConfirm([string]$GivenText, [string]$AccountName) {
    return ($GivenText -ceq ('service-account:' + $AccountName))
}
function Import-DirectoryModule {
    if (-not (Get-Module -ListAvailable -Name ActiveDirectory)) {
        Stop-With 'module_missing' 'The ActiveDirectory PowerShell module is not installed on this computer. Install the RSAT-AD-PowerShell feature.'
    }
    Import-Module ActiveDirectory
}
# The controller this script runs on: every cmdlet addresses it by name, so a change never lands on another one.
function Get-Context([bool]$Write) {
    $dc = Get-ADDomainController -Identity ([System.Net.Dns]::GetHostName())
    $domain = Get-ADDomain -Server ([string]$dc.HostName)
    $forest = Get-ADForest -Server ([string]$dc.HostName)
    $readOnly = ($dc.IsReadOnly -eq $true)
    if ($Write -and $readOnly) { Stop-With 'read_only_dc' 'This is a read-only domain controller; nothing is changed through it.' }
    $rootSid = [string]$domain.DomainSID.Value
    if ($forest.RootDomain -ne $domain.DNSRoot) {
        $rootSid = [string](Get-ADDomain -Identity ([string]$forest.RootDomain) -Server ([string]$dc.HostName)).DomainSID.Value
    }
    $ctxCommon = @{ Server = [string]$dc.HostName }
    $ctxCred = $null
    $hasUser = Test-Param 'credUser'
    $hasPass = Test-Param 'credPassword'
    if ($hasUser -ne $hasPass) { Stop-With 'conflict' 'A credential needs both a user name and a password.' }
    if ($hasUser) {
        $ctxCred = New-Object System.Management.Automation.PSCredential([string]$p.credUser, (ConvertTo-Secure ([string]$p.credPassword)))
        $ctxCommon['Credential'] = $ctxCred
    }
    return @{
        Server = [string]$dc.HostName; Common = $ctxCommon; ReadOnly = $readOnly; Gc = [string]$dc.HostName
        DomainSid = $domain.DomainSID.Value; DomainDn = [string]$domain.DistinguishedName; DomainName = [string]$domain.DNSRoot
        ForestSid = $rootSid; ControllersDn = [string]$domain.DomainControllersContainer
        Forest = $forest; Domain = $domain; Dc = $dc
    }
}
# A SecureString from text without ConvertTo-SecureString -AsPlainText: the text is only ever a variable.
function ConvertTo-Secure([string]$Text) {
    $s = New-Object System.Security.SecureString
    foreach ($ch in $Text.ToCharArray()) { $s.AppendChar($ch) }
    $s.MakeReadOnly()
    return $s
}
function Stop-ForError($err) {
    $t = $err.Exception.GetType().Name
    switch -Regex ($t) {
        'ADIdentityNotFoundException' { Stop-With 'not_found' 'The directory has no such object.' }
        'ADIdentityAlreadyExistsException' { Stop-With 'conflict' 'An object with that name already exists.' }
        'ADPasswordComplexityException|ADInvalidPasswordException|PasswordException' { Stop-With 'password_policy' 'The domain refused the password (length, complexity or history).' }
        'UnauthorizedAccessException|ADException' {
            if ($err.Exception.Message -match 'denied|authoriz|access') { Stop-With 'access_denied' 'The directory refused the change: this computer is not allowed to make it.' }
            Stop-With 'command_failed' ('The directory returned an error: ' + $err.Exception.Message)
        }
        'ADServerDownException|ADWS|ADServerDown' { Stop-With 'adws_unavailable' 'Active Directory Web Services is not running on this controller.' }
        'ADDefaultDomainPasswordPolicy' { Stop-With 'command_failed' ('The directory returned an error: ' + $err.Exception.Message) }
    }
    if ($err.Exception.Message -match 'object not found|does not exist|Cannot find an object') { Stop-With 'not_found' 'The directory has no such object.' }
    if ($err.Exception.Message -match 'Web Services|ADWS|9389') { Stop-With 'adws_unavailable' 'Active Directory Web Services is not running on this controller.' }
    Stop-With 'command_failed' ('The action failed: ' + $t + ': ' + $err.Exception.Message)
}
